// ============================================================= // MAINTENANCE SYSTEM - FINAL VERSION (Custom Job ID & Popups) // ============================================================= const SHEET_NAMES = { REQUESTS: 'Requests', LISTS: 'Lists' }; function doPost(e) { return handleRequest(e); } function doGet(e) { return handleRequest(e); } function handleRequest(e) { var lock = LockService.getScriptLock(); var params = e.parameter || {}; if (e.postData && e.postData.contents) { try { var body = JSON.parse(e.postData.contents); for (var key in body) params[key] = body[key]; } catch(err) {} } if (['submit', 'update', 'manageList'].indexOf(params.action) > -1) { lock.waitLock(10000); } try { var result = {}; var ss = SpreadsheetApp.getActiveSpreadsheet(); if (params.action == 'submit') { result = handleSubmit(ss, params); } else if (params.action == 'fetch') { result = handleFetch(ss); } else if (params.action == 'update') { result = handleUpdate(ss, params); } else if (params.action == 'getLists') { result = handleGetLists(ss); } else if (params.action == 'manageList') { result = handleManageList(ss, params); } else { result = { status: 'error', message: 'Unknown action' }; } var output = JSON.stringify(result); if (params.callback) { return ContentService.createTextOutput(params.callback + '(' + output + ')').setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // --- LIST FUNCTIONS --- function handleGetLists(ss) { var sheet = ss.getSheetByName(SHEET_NAMES.LISTS); if (!sheet) { sheet = ss.insertSheet(SHEET_NAMES.LISTS); sheet.appendRow(["Departments", "Technicians"]); return { status: 'success', data: { depts: [], techs: [] } }; } var lastRow = sheet.getLastRow(); if (lastRow < 2) return { status: 'success', data: { depts: [], techs: [] } }; var data = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); var depts = [], techs = []; for (var i = 0; i < data.length; i++) { if (data[i][0]) depts.push(data[i][0]); if (data[i][1]) techs.push(data[i][1]); } return { status: 'success', data: { depts: depts, techs: techs } }; } function handleManageList(ss, p) { var sheet = ss.getSheetByName(SHEET_NAMES.LISTS); if (!sheet) { sheet = ss.insertSheet(SHEET_NAMES.LISTS); sheet.appendRow(["Departments", "Technicians"]); } var col = p.type === 'depts' ? 1 : 2; if (p.act === 'add') { var allData = sheet.getDataRange().getValues(); var targetRow = 1; for (var r = allData.length; r > 0; r--) { if (allData[r-1][col-1] != "") { targetRow = r; break; } } sheet.getRange(targetRow + 1, col).setValue(p.value); return { status: 'success' }; } else if (p.act === 'del') { var data = sheet.getDataRange().getValues(); for(var i=1; i 1) { var ids = sheet.getRange(2, 1, lastRow-1).getValues().flat(); ids.forEach(function(id) { if (String(id).indexOf(idPrefix) === 0) { var numPart = String(id).replace(idPrefix, ''); var num = parseInt(numPart, 10); if (!isNaN(num) && num > maxId) maxId = num; } }); } // ถ้ายังไม่มี ID ในปีนี้ ให้เริ่มที่ startNum, ถ้ามีแล้วให้ +1 var nextNum = (maxId === 0) ? startNum : (maxId + 1); // Pad with zeros (e.g. 0001) var nextNumStr = nextNum.toString().padStart(4, '0'); var newId = idPrefix + nextNumStr; var imgUrl = ""; if (p.imageFile && p.folderId) { imgUrl = uploadToDrive(p.imageFile, p.folderId, newId + "_BEFORE"); } sheet.appendRow([ newId, new Date(), p.requesterName, p.positionDept, "'"+p.phoneNumber, p.equipmentLocation, p.problemDetails, p.urgency, 'Waiting', '', '', '', '', '', '', '', '', imgUrl, '' ]); return { status: 'success', jobId: newId }; } function handleFetch(ss) { var sheet = ss.getSheetByName(SHEET_NAMES.REQUESTS); if (!sheet || sheet.getLastRow() <= 1) return { status: 'success', data: [] }; var lastRow = sheet.getLastRow(); var lastCol = sheet.getLastColumn(); var data = sheet.getRange(2, 1, lastRow-1, lastCol < 19 ? 19 : lastCol).getValues(); var keys = ['jobID','requestDate','requesterName','positionDept','phoneNumber','equipmentLocation','problemDetails','urgency','status','technician','adminRemarks','acceptedDate','completedDate','cost','itemsData','itemsSummary','additionalNotes','imageBefore','imageAfter']; return { status: 'success', data: data.map(function(r) { var obj = {}; keys.forEach(function(k, i) { var val = (i < r.length) ? r[i] : ""; obj[k] = (val instanceof Date) ? val.toISOString() : (val || ''); }); return obj; }).reverse() }; } function handleUpdate(ss, p) { var sheet = ss.getSheetByName(SHEET_NAMES.REQUESTS); var finder = sheet.getRange("A:A").createTextFinder(p.jobId).matchEntireCell(true).findNext(); if (!finder) return { status: 'error', message: 'Not found' }; var r = finder.getRow(); var set = function(c, v) { if(v!=null) { if (c > sheet.getMaxColumns()) sheet.insertColumnsAfter(sheet.getMaxColumns(), c - sheet.getMaxColumns()); sheet.getRange(r, c).setValue(v); } }; set(9, p.status); set(10, p.technician); set(11, p.adminRemarks); set(17, p.additionalNotes); if(p.status === 'Processing') { var currentDate = sheet.getRange(r, 12).getValue(); if(!currentDate) set(12, new Date()); } if(p.status === 'Done') { set(13, new Date()); set(14, parseFloat(p.cost)); set(15, p.itemsData); set(16, p.itemsSummary); if (p.imageFile && p.folderId) { var imgUrl = uploadToDrive(p.imageFile, p.folderId, p.jobId + "_AFTER"); set(19, imgUrl); } // สร้างรายงาน (ใช้ฟังก์ชันเดิมของคุณที่ผมให้ไปก่อนหน้านี้ได้เลย) // createJobReportSheet(sheet, r); } return { status: 'success' }; } function uploadToDrive(base64Data, folderId, fileName) { try { if(!base64Data || !folderId) return ""; var split = base64Data.split('base64,'); if (split.length < 2) return ""; var contentType = split[0].split(':')[1].split(';')[0]; var bytes = Utilities.base64Decode(split[1]); var blob = Utilities.newBlob(bytes, contentType, fileName); var file = DriveApp.getFolderById(folderId).createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } catch (e) { return "Error: " + e.toString(); } }
